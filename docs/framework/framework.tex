\section{总览}

\begin{itemize}
  \item \textbf{赛道}：自然堆积，破碎骨料 $5$--$40$~mm，单层/准单层撒料，正俯拍静态图像。
  \item \textbf{目标}：精细识别（颗粒与背景/粘连分离）、粒径分析（长短径与 $D_{10}/D_{50}/D_{90}$）、形状分类（针片/圆/棱角/普通）、异常检测（$>50$~mm 异物或泥团），现场采集与调试 $\le30$~分钟，一键本地运行。
  \item \textbf{评分口径}：以标准机械筛分的质量分布为准。本方案在视觉几何与筛分质量语义之间建立显式映射，输出质量代理级配供对标，同时保留数量分布作对照。
  \item \textbf{设计约束}：复用预训练分割能力，不现场重训；流水线可追溯到单颗粒；不引入新的重型依赖。
\end{itemize}

\section{流水线分步说明}

\subsection{分割 — 精细识别}

\begin{itemize}
  \item \textbf{模型}：Cellpose-SAM，权重 \code{IG2_full_set_cp_SAM}（约$1.2$~GB），在骨料全集上预训练，现场不重训。
  \item \textbf{输入输出}：批量读取图像目录，输出 \code{*_pred.tif}（整数掩膜，$0$背景、$1\ldots N$颗粒）与 \code{*_composite.png}（原图与掩膜叠加，供目检粘连与漏检）。
  \item \textbf{关键参数}：\code{--diameter}（目标粒径px，默认自动）、\code{--min_size}（最小面积px$^2$）、\code{--second_diameter}/\code{--comb_threshold}（可选双尺度合并）。
  \item \textbf{质检}：另有 \code{*_detection_overview.png} 三联图（原图｜掩膜｜叠加）。
\end{itemize}

\subsection{几何测量}

\begin{itemize}
  \item \textbf{方法}：对每个标签区做区域属性提取，再拟合椭圆得到长轴$a$、短轴$b$，同时记录面积$A$、凸包面积$A_{\mathrm{convex}}$、周长$P$（Crofton法）、凸度$\mathrm{solidity}=A/A_{\mathrm{convex}}$、方向、质心等。
  \item \textbf{过滤}：剔除贴边（\code{edge}阈值）与过小（\code{px_cutoff}）颗粒，可关闭。
  \item \textbf{产物}：\code{*_pred_grains.csv}，一行一颗粒，含 \code{ell: a-axis (px)}、\code{ell: b-axis (px)}、\code{area}等。
\end{itemize}

\subsection{尺度换算}

\begin{itemize}
  \item \textbf{定义}：\code{resolution}为mm/px，$a/b\,(\mathrm{mm})=a/b\,(\mathrm{px})\times\mathrm{resolution}$，面积仍以px$^2$存储（换算时自乘$\mathrm{resolution}^2$）。
  \item \textbf{获取}：钢尺平放于托盘同平面拍一张，用白尺外框全长$L_{\mathrm{full}}$得$\mathrm{mm/px}=535/L_{\mathrm{full}}$（$500$~mm刻度对应$L_{\text{刻度}}=L_{\mathrm{full}}\times500/535$）；或直接传\code{--resolution 0.208}。输出\code{*_pred_grains_re_scaled.csv}同时含px与mm双列。
\end{itemize}

\subsection{筛分等效粒径}

\begin{itemize}
  \item \textbf{公式}：$d=\theta_1 b+\theta_2 d_{\mathrm{eq}}+\theta_3$，其中$d_{\mathrm{eq}}=2\sqrt{A\cdot\mathrm{resolution}^2/\pi}$为面积等效直径。
  \item \textbf{默认}：$\theta=(1,0,0)$，即$d=b$（短轴），与筛孔通过最相关。
  \item \textbf{校准接口}：有配对筛分真值时可用差分进化拟合$\theta_1,\theta_2\in[0,3],\theta_3\in[-20,20]$mm；本提交固定短轴基线。
\end{itemize}

\subsection{质量加权 — 数量到质量}

\begin{itemize}
  \item \textbf{权重}：$w=d^{\gamma}$，$\gamma=3$。几何相似假设下体积与质量$\propto$尺寸$^3$，归一化后常数消去。
  \item \textbf{分布}：$F(d)=\sum w\cdot\mathbf{1}(d_i\le d)/\sum w$，阶梯逆得$D_{10}/D_{50}/D_{90}$。
  \item \textbf{粒级}：按筛孔$5,10,16,20,25,31.5,40$~mm分桶，含边界桶$<5$与$>40$，质量占比$\%$。
\end{itemize}

\subsection{形状分类}

投影形貌，规则法，阈值可调：

\begin{table}[H]
\centering
\footnotesize
\begin{tabular}{ll}
\toprule
类别 & 判定 \\
\midrule
针片状候选 & $AR=b/a<0.5$ \\
圆形 & $4\pi A/P^2>0.85$ 且 $\mathrm{solidity}>0.95$ 且 $AR>0.8$ \\
棱角状 & $4\pi A/P^2<0.6$ 或 $\mathrm{solidity}<0.9$（且非针片） \\
普通 & 其余 \\
\bottomrule
\end{tabular}
\end{table}

优先级 棱角$>$圆形$>$针片$>$普通。

\subsection{异常检测}

\begin{itemize}
  \item 大块异物：$d>50$~mm
  \item 过小噪声：$d<5$~mm
  \item 疑似泥团：$40<d\le50$且$\mathrm{solidity}<0.85$（可关闭 \code{mud_solidity=None}）
\end{itemize}
判定顺序 \code{oversized} $>$ \code{suspect_mud} $>$ \code{undersized} $>$ \code{normal}。

\subsection{场景汇总与报告}

\begin{itemize}
  \item 输出 \code{SceneSummary} 含 \code{percentiles_number_based} vs \code{percentiles_mass_weighted}、\code{mass_fractions}、\code{normal_only}、\code{shape}与\code{anomalies}。
  \item 提交口径：剔除异常后的正常骨料质量代理分布。
  \item 落盘：\code{_summary.json}、\code{_report.txt}、\code{_particles_annotated.csv}、\code{_gsd_comparison.png}等。
\end{itemize}

\section{边界说明}

\begin{itemize}
  \item 本提交固定短轴基线与$\gamma=3$体积先验，未做筛分真值拟合；拟合接口保留供后续自采数据校准。
  \item 投影形貌为二维规则，非三维量规等价；疑似泥团为几何候选，非材料分类器。
  \item 约$27$~s/张为算法阶段时序，非$30$~min现场全流程实测。
\end{itemize}
