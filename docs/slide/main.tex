\begin{frame}{核心方法 1：把二维几何压缩为“筛分等效粒径”}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{block}{为什么不能直接把图像短轴当作机械筛分真值？}
\small 机械筛分观察多姿态筛孔通过；单目图像只给当前视角的二维投影。两者相关，但并不等价。
\end{block}
\vspace{0.4em}
\centering
{\Large
\[
 d_i=\theta_1 b_i+\theta_2 d_{\mathrm{eq},i}+\theta_3
\]
}
\vspace{-0.4em}
\begin{itemize}\footnotesize
  \item $b_i$：投影窄向尺度，与筛孔通过直接相关；
  \item $d_{\mathrm{eq},i}$：补充整体投影面积信息；
  \item 低维参数保持可解释、可校准、可复算。
\end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
\begin{alertblock}{本文固定基线}
\centering
{\Large $\boldsymbol{\theta}=(1,0,0)$}\\[0.4em]
\small 即 $d_i=b_i$，所有场景使用同一组参数，不根据输出级配临场调参。
\end{alertblock}
\vspace{0.4em}
\smallgray{代码保留外部机械筛分数据下的低维校准接口；当前三张演示图没有配对逐筛称量，因此本文不执行该拟合。}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{核心方法 2：数量统计 $\neq$ 质量筛分，必须显式建模权重}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{视觉天然按“颗粒个数”观察}
\centering
\begin{tikzpicture}[scale=0.68]
\foreach \x/\r in {0/0.22,0.65/0.20,1.25/0.23,1.85/0.19,2.45/0.21,3.2/0.48}{
  \draw[fill=blue!10] (\x,0) circle (\r);
}
\node[font=\scriptsize,align=center] at (1.6,-0.72) {6 个实例\\每个实例等权};
\end{tikzpicture}
\end{block}
\footnotesize
\textbf{机械筛分按质量统计：}粗颗粒数量少，但单颗粒质量贡献通常远大于细颗粒。
\end{column}
\begin{column}{0.51\textwidth}
\centering
{\large $w_i=d_i^{\gamma},\qquad \gamma=3$}
\vspace{0.15em}
\smallgray{$\gamma=3$ 是几何相似条件下的体积/质量尺度先验，不是机械称量校准结果。}
\vspace{0.2em}

{\small\[
F(d)=\frac{\sum_i w_i\mathbf{1}(d_i\le d)}{\sum_i w_i}
\]
\[
D_p=\inf\{d:F(d)\ge p/100\}
\]}
\vspace{0.1em}
\begin{alertblock}{这一转换是整套方案的统计核心}
\footnotesize 同一实例集合只改变权重，就能直接观察“数量口径”和“质量代理口径”的差异。
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{关键结果：同一实例集合上，质量代理显著改变分布中心}

\begin{tabular}{lrrr}

\bottomrule
\end{tabular}}
\vspace{0.55em}
\smallgray{单位：mm；使用全部检测实例，不做异常剔除，只把等权切换为 $d^3$ 权重。}
\end{column}
\begin{column}{0.42\textwidth}
\begin{alertblock}{核心观察}
\small 三个场景都向粗端移动：\textbf{数量分布不能直接替代质量相关级配。}
\end{alertblock}
\vspace{0.25em}
\begin{block}{机制解释}
\footnotesize
\begin{itemize}
  \item 小颗粒实例数量多，但单颗粒质量贡献较低；
  \item 粗颗粒数量较少，在 $d^3$ 权重下统计贡献显著增大；
  \item merge 形成的伪粗颗粒会被幂次权重进一步放大。
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}
